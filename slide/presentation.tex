%% Single pill — INLINE TikZ (no remember picture, no overlay)
%% Usage: \pill{num}{TEXT}
\newcommand{\pill}[2]{%
  \begin{tikzpicture}[baseline=0.1ex]%
    \fill[pillblue, rounded corners=6pt]
      (0,-0.38cm) rectangle (4.4cm,0.38cm);
    \fill[midblue, rounded corners=6pt]
      (0,-0.38cm) rectangle (0.88cm,0.38cm);
    \fill[deepblue]
      (0.44cm,0) circle (0.31cm);
    \node[text=white, font=\fontsize{7}{7}\selectfont\bfseries]
      at (0.44cm,0) {#2};
    \node[text=white, font=\fontsize{6.0}{7}\selectfont\bfseries,
          anchor=west, text width=3.2cm, align=left]
      at (0.88cm,0) {#1};%
  \end{tikzpicture}%
}
